def find_nn_slice(slices, nn_distance):  # slices contains polygons of two adjacent slices
    neighbors = np.zeros(shape=(len((slices - 1) / 2), slices[0].p + 1), dtype=np.int32)
    col, row = 1, 0
    for polygon in slices:
        if polygon.sector == 0:
            for pgon in slices:
                dist = weierstrass_distance(pgon.centerW, polygon.centerW)
                if polygon.centerP != pgon.centerP and round(dist, 10) == round(nn_distance, 10):
                    neighbors[row, col] = pgon.number
                    col += 1
            row += 1
            col = 1
